The multiplication is performed using three nested `for` loops. For square matrices, the time complexity is **O(n³)** because the calculation depends on three loops over the matrix size.

I used the same matrix dimensions and values as in the NumPy experiment. The C++ program produced the same result:

#### C++ — Implementation, Unit Tests and Benchmark

```cpp
#include <iostream>
#include <vector>
#include <cassert>
#include <chrono>
#include <random>

using Matrix = std::vector<std::vector<double>>;

Matrix multiply(const Matrix& A, const Matrix& B) {

    size_t rows = A.size();
    size_t common = A[0].size();
    size_t cols = B[0].size();

    Matrix C(
        rows,
        std::vector<double>(cols, 0.0)
    );

    for (size_t i = 0; i < rows; i++) {

        for (size_t k = 0; k < common; k++) {

            for (size_t j = 0; j < cols; j++) {

                C[i][j] += A[i][k] * B[k][j];
            }
        }
    }

    return C;
}


void test_basic() {

    Matrix A = {
        {1, 2},
        {3, 4}
    };

    Matrix B = {
        {5, 6},
        {7, 8}
    };

    Matrix expected = {
        {19, 22},
        {43, 50}
    };

    Matrix result = multiply(A, B);

    assert(result == expected);

    std::cout << "Test 1 passed: basic multiplication\n";
}


void test_rectangular() {

    Matrix A = {
        {1, 2, 3},
        {4, 5, 6}
    };

    Matrix B = {
        {7, 8},
        {9, 10},
        {11, 12}
    };

    Matrix expected = {
        {58, 64},
        {139, 154}
    };

    Matrix result = multiply(A, B);

    assert(result == expected);

    std::cout << "Test 2 passed: rectangular matrices\n";
}


void test_identity() {

    Matrix A = {
        {2, 3},
        {4, 5}
    };

    Matrix identity = {
        {1, 0},
        {0, 1}
    };

    Matrix result = multiply(A, identity);

    assert(result == A);

    std::cout << "Test 3 passed: identity matrix\n";
}


void test_zero_matrix() {

    Matrix A = {
        {1, 2},
        {3, 4}
    };

    Matrix zero = {
        {0, 0},
        {0, 0}
    };

    Matrix expected = {
        {0, 0},
        {0, 0}
    };

    Matrix result = multiply(A, zero);

    assert(result == expected);

    std::cout << "Test 4 passed: zero matrix\n";
}


Matrix generate_matrix(size_t n) {

    Matrix matrix(
        n,
        std::vector<double>(n)
    );

    std::random_device rd;
    std::mt19937 generator(rd());

    std::uniform_real_distribution<double> distribution(
        0.0,
        1.0
    );

    for (size_t i = 0; i < n; i++) {

        for (size_t j = 0; j < n; j++) {

            matrix[i][j] = distribution(generator);
        }
    }

    return matrix;
}


double benchmark(size_t n, int repetitions = 5) {

    Matrix A = generate_matrix(n);
    Matrix B = generate_matrix(n);

    double best_time = 1e100;

    for (int r = 0; r < repetitions; r++) {

        auto start =
            std::chrono::steady_clock::now();

        Matrix C = multiply(A, B);

        auto end =
            std::chrono::steady_clock::now();

        double elapsed =
            std::chrono::duration<double>(
                end - start
            ).count();

        if (elapsed < best_time) {
            best_time = elapsed;
        }
    }

    return best_time;
}


void print_matrix(const Matrix& matrix) {

    for (const auto& row : matrix) {

        for (double value : row) {
            std::cout << value << " ";
        }

        std::cout << "\n";
    }
}


int main() {

    std::cout << "Running unit tests...\n\n";

    test_basic();
    test_rectangular();
    test_identity();
    test_zero_matrix();

    std::cout << "\nAll unit tests passed!\n";

    Matrix A = {
        {1, 2},
        {3, 4}
    };

    Matrix B = {
        {5, 6},
        {7, 8}
    };

    Matrix C = multiply(A, B);

    std::cout << "\nMatrix multiplication example:\n";

    print_matrix(C);

    std::cout << "\nC++ execution times:\n";

    for (size_t n : {100, 300, 500}) {

        double time_taken = benchmark(n);

        std::cout
            << n << " x " << n
            << ": "
            << time_taken
            << " seconds\n";
    }

    return 0;
}
```
